order_details = []
total_amount = 0

def add_order(product_name , quantity , price):
    order_details.append(
        {
            "product_name" :  product_name,
            "product_count" : quantity,
            "price"         : price
        }
    )

def calculate_total_bill_amount():
    total_amount = 0
    for i in order_details:
        total_amount += i.get("price") * i.get("product_count")
    return total_amount

def print_bill():
    for i in order_details:
        print(f" Product :: {i.get('product_name') }:: {i.get('product_count')} :: ({i.get('price') * i.get('product_count')}))")
    total_bill_amount = calculate_total_bill_amount()
    print("Total amount is :: ",total_bill_amount)

while True:
