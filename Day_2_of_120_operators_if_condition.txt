and , or

num1 = 10
num2 = 3

# Arithmetic operators
print ("Sum is", num1 + num2 )
print(f"Sum is {num1 + num2}")
print(f"Difference is {num1 - num2}")
print(f"Multiplication is {num1 * num2}")
print(f"Division is {num1 / num2}")
print(f"Quotient is {num1 // num2}")
print(f"Reminder is {num1 % num2}")
print(f"Power is {num1 ** num2}")
print ( 12 / 4 )

# Comparison operators
print(f"Is equal? {num1 == num2}")
print(f"Is Less than? {num1 < num2}")
print(f"Is Less than or equal? {num1 <= num2}")
print(f"Is not equal? {num1 != num2}")
print(f"Is Greater than? {num1 > num2}")
print(f"Is Greater than or equal? {num1 >= num2}")

# Logical operators

print( num1 == num2 and num1 != num2 )
print( num1 == num2 or num1 != num2 )

    print("Num1 is greater than Num2")
