marks = [100 , 96 , 66 , 89 , 75]

print ( f" Minimum Marks :: {min(marks)}")  # Maximum value
print ( f" Maximum Marks :: {max(marks)}")  # Minimum Value
print ( f" Total Marks :: {sum(marks)}")    # Total Value

marks.sort()   # It sorts the list in ascending order
print(marks)

marks.sort(reverse = True)  # It sorts the list in descending order
print(marks)

marks.remove(66)  # It is used to remove the values in the list using the value
print(marks)

marks.pop(2) # It is used to remove the values in the list using the index position
print(marks)

marks.append(80) # It is used to assign the new value to the list
print(marks)

students.extend(marks) # It is used to combine two lists into a single list
print(students)

marks.clear()  # It is used to empty the list
print(marks)
