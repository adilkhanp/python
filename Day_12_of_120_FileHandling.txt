##### FILE HANDLING ############


w -- write mode .. it clears old content and writes new content
a -- append mode.. it writes to the existing content
r -- read mode... it is used to read the data from the file

file = open("SampleFile.txt",'w')
file = open("SampleFile.txt",'a')
file.write("This is my fourth line from python")
file.write("This is my fifth line from python")
file.write("This is my sixth line from python")

file.close()

file = open("SampleFile.txt",'r')
dat = file.read()  #### It reads complete file data at once
print(dat)

dat = file.readlines()
